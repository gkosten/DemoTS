```dataviewjs
// ============================================================
// НАСТРОЙКИ
// ============================================================
const folderName = "DemoTS";
const minHeadingLevel = 1;
const maxHeadingLevel = 3;
const showFolderHeaders = true;

// ⬇️ СПИСОК ИСКЛЮЧЕНИЙ
// Сюда вписывайте имена файлов (с .md или без) и имена папок,
// которые не должны попадать в оглавление.
// Регистр не важен. Совпадение по имени — точное, не по подстроке.
const excludeList = [
    // Примеры — замените на свои:
    "Оглавление.md",
    "README",
    "README.md",
    "Черновик",
    "Черновик.md",
    "drafts",       // папка
    "_private",     // папка
    "assets",       // папка с картинками
    ".obsidian",    // служебная папка Obsidian
];

// Приводим список к нижнему регистру один раз — для быстрой проверки
const excludeSet = new Set(excludeList.map(s => s.toLowerCase()));

// ============================================================
// СБОР ФАЙЛОВ
// ============================================================
const allFiles = app.vault.getMarkdownFiles();

const files = allFiles
    .filter(f => {
        const path = f.path;
        return path.startsWith(folderName + "/")
            || path.includes("/" + folderName + "/");
    })
    .filter(f => {
        // Проверяем имя файла (с расширением и без)
        const fileNameWithExt = f.name.toLowerCase();      // "index.md"
        const fileNameNoExt  = f.basename.toLowerCase();   // "index"

        if (excludeSet.has(fileNameWithExt) || excludeSet.has(fileNameNoExt)) {
            return false;
        }

        // Проверяем все папки в пути
        const parts = f.path.split("/");
        for (const part of parts) {
            if (excludeSet.has(part.toLowerCase())) {
                return false;
            }
        }

        return true;
    })
    .sort((a, b) =>
        a.path.localeCompare(b.path, undefined, { numeric: true })
    );

// ============================================================
// ВЫВОД
// ============================================================
if (files.length === 0) {
    dv.paragraph(`*Не найдено файлов в папке "${folderName}"*`);
} else {
    let currentSubfolder = null;

    for (const file of files) {
        const parts = file.path.split("/");
        const rootIndex = parts.indexOf(folderName);
        const subPath = parts.slice(rootIndex + 1, -1).join("/");

        if (showFolderHeaders && subPath !== currentSubfolder) {
            currentSubfolder = subPath;
            if (subPath) {
                dv.header(4, `📁 ${subPath}`);
            }
        }

        const cache = app.metadataCache.getFileCache(file);
        const link = dv.fileLink(file.path);

        dv.header(5, link);

        if (cache && cache.headings) {
            const headings = cache.headings
                .filter(h => h.level >= minHeadingLevel && h.level <= maxHeadingLevel);

            if (headings.length > 0) {
                const lines = headings.map(h => {
                    const indent = '  '.repeat(Math.max(0, h.level - minHeadingLevel));
                    return `${indent}- [[${file.basename}#${h.heading}|${h.heading}]]`;
                });
                dv.paragraph(lines.join('\n'));
            } else {
                dv.paragraph(`  *Нет заголовков H${minHeadingLevel}-H${maxHeadingLevel}*`);
            }
        }
    }
}
```
